// LAUNCH | builds, creates, starts, attaches --------------------------------------------
docker-compose up
// LAUNCH | choose docker-compose.yml file 
docker-compose -f <filename> up

// LAUNCH | IN BACKGROUND 
docker-compose up -d

// NO START CONTAINER 
docker-compose up --no-start

// LAUNCH | REBUILD IMAGE 
docker-compose up --build 

// SPECIFIC CONTAINER
docker-compose up <service name>
docker-compose up -d --build <service name>
docker-compose -f <file name> up -d --build
docker-compose start <service name>
docker-compose stop <service name>
docker-compose logs <service name>

// STOP CONTAINERS ---------------------------------------------------------------
docker-compose down

// LISTING RUNNING CONTAINERS ----------------------------------------------------
docker-compose ps

// VALIDATE docker-compose.yml file ----------------------------------------------
docker-compose config
docker-compose config -q

// PULL IMAGES in docker-compose.yml file ----------------------------------------
docker-compose pull

// BUILD IMAGES in docker-compose.yml file, no configure the container -----------
docker-compose build

// BUILD IMAGES in docker-compose.yml file ---------------------------------------
docker-compose create

// START, STOP, RESTART, PAUSE, UNPAUSE ------------------------------------------
docker-compose start
docker-compose stop
docker-compose restart
docker-compose pause
docker-compose unpause

// TOP, LOGS, EVENTS ------------------------------------------------------------
docker-compose top
docker-compose top <service name>
docker-compose logs -f 
docker-compose events

// EXEC AND RUN -----------------------------------------------------------------
docker-compose exec <service name> <command>
docker-compose exec worker ping -c 3 db

// KILL, RM, DOWN ---------------------------------------------------------------
docker-compose kill

// REMOVES ANY CONTAINERS WITH THE STATE OF EXITED ------------------------------
docker-compose rm

// REMOVES ALL OF THE CONTAINERS, NETWORKS, VOLUMES, IMAGES ---------------------
docker-compose down --rmi all --volumes
