// Connect a container to a network ----------------------------------------------
docker network connect <network> <container id>

// Create a network --------------------------------------------------------------
docker network create <network>

// Disconnect a container from a network -----------------------------------------
docker network disconnect <network> <container id>

// Display detailed information on one or more networks --------------------------
docker network inspect <network>

// List networks -----------------------------------------------------------------
docker network ls

// Remove all unused networks ----------------------------------------------------
docker network prune

// Remove one or more networks ---------------------------------------------------
docker network rm <network>

// EXAMPLE
docker image pull redis:alpine
docker image pull russmckendrick/moby-counter
docker network create moby-counter
docker run -d --name redis --network moby-counter redis:alpine
docker run -d --name moby-counter --network moby-counter -p 8080:80 russmckendrick/moby-counter

// EXEC CHECK NETWORK
docker exec moby-counter ping -c 3 redis
docker exec moby-counter cat /etc/hosts
docker exec moby-counter cat /etc/resolv.conf
docker exec moby-counter nslookup redis 127.0.0.11

docker network create moby-counter2
docker run -itd --name moby-counter2 --network moby-counter2 -p 9090:80 russmckendrick/moby-counter
docker container exec moby-counter2 ping -c 3 redis
docker container exec moby-counter2 cat /etc/resolv.conf
docker container exec moby-counter2 nslookup redis 127.0.0.11
docker container run -d --name redis2 --network moby-counter2 --network-alias redis redis:alpine
docker network inspect moby-counter

docker container stop moby-counter2 redis2
docker container prune
docker network prune
