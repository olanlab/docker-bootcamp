# DEFAULT 
docker run -d --name my-nginx -p 8080:80  nginx

# MAPPING | WEBSITE
docker run -d -p 8081:80 --name my-nginx -v $(pwd)/demo:/usr/share/nginx/html nginx

# MAPPING | nginx.conf
docker run -d -p 8082:80 --name my-nginx -v $(pwd)/demo:/usr/share/nginx/html -v $(pwd)/conf/nginx.conf:/etc/nginx/nginx.conf nginx
docker run -d -p 8082:80 --name my-nginx -v $(pwd)/web:/usr/share/nginx/html -v $(pwd)/conf/nginx.conf:/etc/nginx/nginx.conf nginx

# MAPPING | PHP
docker run -d -p 8082:80 --name my-nginx -v $(pwd)/web:/usr/share/nginx/html -v $(pwd)/conf/nginx.conf:/etc/nginx/nginx.conf olanlab/nginx-phpfpm

# MAPPING | https
docker run -d -p 443:443 --name my-nginx -v $(pwd)/web:/usr/share/nginx/html -v $(pwd)/conf/nginx.conf:/etc/nginx/nginx.conf olanlab/nginx-phpfpm

# STOP & REMOVE
docker stop my-nginx && docker rm my-nginx

# EXEC
docker exec -it my-nginx bash
------------------------------------------------------------------------------------------
# NGINX 
service nginx reload

# INSTALL NANO, VIM
apt-get update
apt-get install nano vim

------------------------------------------------------------------------------------------
# INSTALL | PHP-FPM
apt-get update
apt-get install php-fpm
# SERVICE | PHP-FPM
service php7.0-fpm start
service php7.0-fpm stop
service php7.0-fpm restart
service php7.0-fpm status

# INSTALL | php7.0-mysql
apt-get install php7.0-mysql
service php7.0-fpm restart

# INSTALL | curl
apt-get install curl
curl -I http://localhost/images/thumb.png
curl -I  http://localhost/style.css
curl -I -H "Accept-Encoding: gzip, deflate" http://localhost/style.css
curl http://localhost/style.css
curl -H "Accept-Encoding: gzip, deflate" http://localhost/style.css
------------------------------------------------------------------------------------------
# INSTALL | apache2-utils
apt-get install apache2-utils

# CREATE | .htpasswd
htpasswd -c /etc/nginx/.htpasswd <username>

------------------------------------------------------------------------------------------
# CREATE | the SSL Certificate
openssl req -x509 -nodes -days 365 -newkey rsa:2048 -keyout /etc/nginx/ssl/self.key -out /etc/nginx/ssl/self.crt

# CREATE | dhparam.pem
openssl dhparam 2048 -out /etc/nginx/ssl/dhparam.pem
openssl dhparam -rand - 2048 > /etc/nginx/ssl/dhparam.pem

# RUN PHP SERVER 
php -S localhost:<port>
php -S localhost:<port> <file>
php -S localhost:9999 index.php

# LOOP ON LINUX
while sleep 0.5; do curl -I https://www.google.com; done
