# DEFAULT 
docker run -d --name my-nginx -p 8080:80  nginx

# MAPPING | WEBSITE
docker run -d -p 8081:80 --name my-nginx -v $(pwd)/demo:/usr/share/nginx/html nginx

# MAPPING | nginx.conf
docker run -d -p 8082:80 --name my-nginx -v $(pwd)/demo:/usr/share/nginx/html -v $(pwd)/conf/nginx.conf:/etc/nginx/nginx.conf nginx
docker run -d -p 8082:80 --name my-nginx -v $(pwd)/web:/usr/share/nginx/html -v $(pwd)/conf/nginx.conf:/etc/nginx/nginx.conf nginx

# STOP & REMOVE
docker stop my-nginx && docker rm my-nginx

# EXEC
docker exec -it my-nginx bash
